# LearnFlow backend runtime dependencies.
#
# Direct dependencies only. Transitive packages are resolved by pip and are not

fastapi==0.140.0
pydantic-settings==2.14.2
uvicorn==0.51.0

# Persistence. SQLAlchemy maps infrastructure models to PostgreSQL and Alembic
# versions the schema; psycopg 3 is the driver. The synchronous execution model
# is recorded in docs/adr/ADR-011-sqlalchemy-persistence-implementation.md.
#
# psycopg[binary] ships prebuilt wheels, so no local libpq or C toolchain is
# needed on a contributor machine or in the container build.
SQLAlchemy==2.0.51
alembic==1.18.5
psycopg[binary]==3.3.4

# Learner-uploaded PDF files (RES-014 to RES-017), per
# docs/adr/ADR-040-learner-uploaded-resource-files.md.
#
# python-multipart is what lets FastAPI parse an uploaded file at all; without
# it a multipart request cannot be read. pypdf is pure Python -- no system
# libraries and no C toolchain -- and is used to count pages and detect an
# encrypted document. It is never asked to extract text: this feature stores
# files and reads them back, and extraction is a separate, unbuilt capability.
python-multipart==0.0.20
pypdf==6.1.3

# What lets pypdf read an AES-encrypted PDF at all. It is optional to pypdf, and
# without it an AES document raises `DependencyError` from inside the parser
# rather than reporting whether it is readable.
#
# It matters because most encrypted study material is not password-locked: a
# publisher or scanned PDF is commonly encrypted with an **empty** user password
# and carries only permission restrictions. Those open in any reader, and
# LearnFlow can read and store them; a document that genuinely needs a password
# still cannot be opened and is refused. Distinguishing the two needs this.
#
# Ships prebuilt wheels, so no local C toolchain is required.
cryptography==46.0.4
